  <script>

    const riseInput = document.getElementById('riseInput');

    const ratioBtns = document.querySelectorAll('.ratio-btn');


    const outRun = document.getElementById('outRun');

    const outRatio = document.getElementById('outRatio');

    const outGrade = document.getElementById('outGrade');

    const outAngle = document.getElementById('outAngle');


    // 결과 항목(4개)

    const resultItems = document.querySelectorAll('.result-item');

    const itemRun = resultItems[0];

    const itemRatio = resultItems[1];

    const itemGrade = resultItems[2];

    const itemAngle = resultItems[3];


    // 모드 버튼

    const modeBtns = document.querySelectorAll('.mode-btn');


    let currentRatio = 12;

    let currentMode = 'arch'; // arch | civil | eng


    function setPrimary(item) {

      [itemRun, itemRatio, itemGrade, itemAngle].forEach(i => {

        i.classList.remove('primary-metric', 'secondary-metric');

        i.classList.add('secondary-metric');

      });

      item.classList.remove('secondary-metric');

      item.classList.add('primary-metric');

    }


    function applyMode() {

      if (currentMode === 'arch') setPrimary(itemRatio);       // 건축: 1:n

      else if (currentMode === 'civil') setPrimary(itemGrade); // 토목: %

      else setPrimary(itemAngle);                               // 엔지: °

    }


    function calculate() {

      const riseCm = parseFloat(riseInput.value) || 0;


      const runCm = riseCm * currentRatio;

      const runM = runCm / 100;

      const gradePercent = 100 / currentRatio;

      const angleDeg = Math.atan(1 / currentRatio) * (180 / Math.PI);


      outRun.textContent = runM.toFixed(2);

      outRatio.textContent = `1:${currentRatio}`;

      outGrade.textContent = gradePercent.toFixed(2);

      outAngle.textContent = angleDeg.toFixed(2);


      applyMode();

    }


    riseInput.addEventListener('input', calculate);


    ratioBtns.forEach(btn => {

      btn.addEventListener('click', (e) => {

        ratioBtns.forEach(b => b.classList.remove('active'));

        e.currentTarget.classList.add('active');

        currentRatio = parseInt(e.currentTarget.dataset.ratio, 10);

        calculate();

      });

    });


    modeBtns.forEach(btn => {

      btn.addEventListener('click', (e) => {

        modeBtns.forEach(b => b.classList.remove('active'));

        e.currentTarget.classList.add('active');

        currentMode = e.currentTarget.dataset.mode;

        calculate();

      });

    });


    calculate();

  </script>
